// Isolated test-data initializer; copied under backend/cmd for internal package access.
package main

import (
	"ant-chrome/backend/internal/config"
	"ant-chrome/backend/internal/database"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strconv"
)

func main() {
	if len(os.Args)!=4 && len(os.Args)!=5 { panic("usage: seed ROOT CHROMIUM PORT [LOCAL_START_URL]") }
	root, err := filepath.Abs(os.Args[1]); if err!=nil { panic(err) }
	if _,err=os.Stat(filepath.Join(root,".linux-perf-runtime")); err!=nil { panic(err) }
	if err=os.MkdirAll(filepath.Join(root,"data"),0700); err!=nil { panic(err) }
	cfg:=config.DefaultConfig()
	cfg.LaunchServer.Port,err=strconv.Atoi(os.Args[3]); if err!=nil { panic(err) }
	cfg.LaunchServer.Auth.Enabled=true
	cfg.LaunchServer.Auth.APIKey="synthetic-local-performance-fixture"
	cfg.Browser.ChromeBinaryPath=os.Args[2]
	cfg.Browser.DefaultCoreId="perf-chromium"
	cfg.Browser.Cores=[]config.BrowserCore{{CoreId:"perf-chromium",CoreName:"Performance Chromium",CorePath:os.Args[2],IsDefault:true}}
	cfg.Browser.UserDataRoot="data/profiles"
	if len(os.Args)==5 { cfg.Browser.DefaultStartURLs=[]string{os.Args[4]} }
	if err=cfg.Save(filepath.Join(root,"config.yaml"));err!=nil {panic(err)}
	db,err:=database.NewDB(filepath.Join(root,cfg.Database.SQLite.Path));if err!=nil{panic(err)}
	defer db.Close()
	if err=db.Migrate();err!=nil{panic(err)}
	json.NewEncoder(os.Stdout).Encode(map[string]string{"root":root,"database":filepath.Join(root,cfg.Database.SQLite.Path),"status":"ready"})
	fmt.Fprintln(os.Stderr,"Synthetic performance database initialized")
}
